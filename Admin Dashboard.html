const SS_ID = '1sUWr2hjU2taHlT5PZL4Lut8IZYgAIWD_i3mVUGgE3YU';

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.openById(SS_ID);
    const RS = body.regSheet || 'Registrations';
    const PS = body.paySheet || 'Payments';

    if (body.action === 'addRegistration') {
      const d = body.data;
      ss.getSheetByName(RS).appendRow([
        new Date().toISOString(), d.regNo, d.name,
        d.phone, d.notes||'', d.course, d.amount, '', d.trainer, d.id
      ]);
      return ok();
    }
    if (body.action === 'updateInvoiceNo') {
      const sheet = ss.getSheetByName(RS);
      const vals = sheet.getDataRange().getValues();
      for (let i = 1; i < vals.length; i++) {
        if (vals[i][9] == body.data.hash) {
          sheet.getRange(i+1, 8).setValue(body.data.invNo); break;
        }
      }
      return ok();
    }
    if (body.action === 'addPayment') {
      const d = body.data;
      ss.getSheetByName(PS).appendRow([
        new Date().toISOString(), d.invNo, d.regNo, d.invNo,
        d.course, d.amount, d.paymentMode, d.name, d.trainer, d.notes||''
      ]);
      return ok();
    }
    if (body.action === 'deleteRegistration') {
      const sheet = ss.getSheetByName(RS);
      const vals = sheet.getDataRange().getValues();
      for (let i = 1; i < vals.length; i++) {
        if (vals[i][9] == body.data.hash) { sheet.deleteRow(i+1); break; }
      }
      return ok();
    }
    return err('Unknown action');
  } catch(e) { return err(e.toString()); }
}

function doGet(e) {
  try {
    const ss = SpreadsheetApp.openById(SS_ID);
    const RS = e.parameter.regSheet || 'Registrations';
    const PS = e.parameter.paySheet || 'Payments';
    const rSheet = ss.getSheetByName(RS);
    const pSheet = ss.getSheetByName(PS);
    return ContentService.createTextOutput(JSON.stringify({
      ok: true,
      regs: rSheet ? rSheet.getDataRange().getValues() : [],
      pays: pSheet ? pSheet.getDataRange().getValues() : []
    })).setMimeType(ContentService.MimeType.JSON);
  } catch(e) { return err(e.toString()); }
}

function ok(d){ return ContentService.createTextOutput(
  JSON.stringify({ok:true,...(d||{})})).setMimeType(ContentService.MimeType.JSON); }
function err(m){ return ContentService.createTextOutput(
  JSON.stringify({ok:false,error:m})).setMimeType(ContentService.MimeType.JSON); }
